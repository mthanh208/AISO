#!/usr/bin/env bash
# =====================================================================
# pet.txt — Consolidated setup & verify wrapper for AISO repository
# Purpose: provide a clean entrypoint that prepares the ai_stack, runs
# basic verification (file presence + python syntax checks), and
# instructs the user how to continue. This file is intended to be
# executable by the repo consumer (it does not run here on GitHub).
# =====================================================================
set -euo pipefail

C_CYAN='\033[96m'; C_GREEN='\033[92m'; C_YELLOW='\033[93m'; C_RED='\033[91m'; C_RESET='\033[0m'
info(){ echo -e "${C_CYAN}$*${C_RESET}"; }
ok(){ echo -e "${C_GREEN}$*${C_RESET}"; }
warn(){ echo -e "${C_YELLOW}$*${C_RESET}"; }
err(){ echo -e "${C_RED}$*${C_RESET}"; }

ROOT_DIR="$(cd "$(dirname "$0")" && pwd)"
AI_DIR="$ROOT_DIR/ai_stack"

info "AISO — setup wrapper"

# 1) Ensure Python present
if ! command -v python3 >/dev/null 2>&1; then
    err "Python3 is required. Please install python3 and rerun."
    exit 1
fi
PYVER=$(python3 -c 'import sys; print(".".join(map(str, sys.version_info[:2])))')
ok "Found Python ${PYVER}"

# 2) Make build script executable if present
if [ -f "$ROOT_DIR/build_ai_stack.sh" ]; then
    chmod +x "$ROOT_DIR/build_ai_stack.sh"
    info "Running build_ai_stack.sh to create ai_stack/..."
    bash "$ROOT_DIR/build_ai_stack.sh"
else
    warn "build_ai_stack.sh not found — skipping ai_stack generation"
fi

# 3) Quick verification steps (file presence + python syntax checks)
MISSING=0
REQUIRED_FILES=(
    "$AI_DIR/core/rule_engine.py"
    "$AI_DIR/core/symbolic_engine.py"
    "$AI_DIR/core/state_machine.py"
    "$AI_DIR/core/ml_utils.py"
    "$AI_DIR/core/ner_spacy.py"
    "$AI_DIR/core/sql_memory.py"
    "$AI_DIR/core/vector_index.py"
    "$AI_DIR/core/scheduler.py"
    "$AI_DIR/core/workflow_orchestrator.py"
    "$AI_DIR/core/sandbox.py"
)

info "Verifying required files in ${AI_DIR}..."
for f in "${REQUIRED_FILES[@]}"; do
    if [ ! -f "$f" ]; then
        warn "  MISSING: $f"
        MISSING=$((MISSING+1))
    else
        ok "  OK: $f"
    fi
done

if [ "$MISSING" -ne 0 ]; then
    warn "Some required files are missing ($MISSING). Please run build_ai_stack.sh or check repository." 
else
    ok "All required files present."
fi

# 4) Python syntax check (compile) for core modules we added
info "Running python -m py_compile on core modules (syntax check)..."
PY_ERRORS=0
for pyf in "$AI_DIR"/core/*.py; do
    if [ -f "$pyf" ]; then
        if ! python3 -m py_compile "$pyf" 2>/tmp/pet_py_errs; then
            err "Syntax error in $pyf"
            sed -n '1,200p' /tmp/pet_py_errs || true
            PY_ERRORS=$((PY_ERRORS+1))
        else
            ok "Compiled: $(basename "$pyf")"
        fi
    fi
done
rm -f /tmp/pet_py_errs || true

if [ "$PY_ERRORS" -ne 0 ]; then
    err "Python syntax checks failed: $PY_ERRORS files. Fix before running the Pet." 
else
    ok "Python syntax checks passed."
fi

# 5) Quick pip install advice
info "If you want to install dependencies now, run:"
cat <<'HINT'
  cd ai_stack
  python -m venv .venv
  source .venv/bin/activate
  pip install -r requirements.txt

Notes:
- faiss-cpu may require platform-specific installation; if pip install fails,
  consider conda: conda install -c pytorch faiss-cpu
- For spaCy models (English): python -m spacy download en_core_web_sm
HINT

# 6) Final status
if [ "$MISSING" -eq 0 ] && [ "$PY_ERRORS" -eq 0 ]; then
    ok "VERIFIED: repository ai_stack is present and core modules compile. You can clone and proceed."
    exit 0
else
    warn "VERIFICATION: issues detected. See messages above."
    exit 2
fi
